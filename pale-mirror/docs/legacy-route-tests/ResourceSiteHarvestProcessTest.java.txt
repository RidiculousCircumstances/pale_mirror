    @Test
    void explicitFresh65CellFieldAdmitsOneBoundedWorkerSegmentAndRetainsItsGenesisAcrossRestart() {
        SubjectId siteId = new SubjectId("site:1-wheat-field");
        FrontierBootstrap baseline = FrontierBootstrapper.create(new WorldId("frontier:resource-site-harvest-65"), 125L);
        ResourceFieldLayout original = FrontierResourceSitePlan.compile(baseline).get(siteId).layout();
        FrontierBootstrap authored = null;
        ResourceFieldLayout extended = null;
        BlockPosition last = original.cells().getLast().crop();
        for (BlockPosition crop : List.of(last.offset(1, 0, 0), last.offset(-1, 0, 0),
                last.offset(0, 0, 1), last.offset(0, 0, -1))) {
            if (original.contains(crop) || original.contains(crop.offset(0, -1, 0))) continue;
            var cells = new java.util.ArrayList<>(original.cells());
            var soil = new SurfaceAnchor(crop.offset(0, -1, 0));
            cells.add(new ResourceFieldLayout.Cell(new ResourceFieldLayout.CellId(original.nextCellId()), crop, soil, soil));
            ResourceFieldLayout candidate = new ResourceFieldLayout(1L, original.nextCellId() + 1L,
                    cells, original.irrigationSlots());
            FrontierBootstrap proposed = new FrontierBootstrap(baseline.worldId(), baseline.seed(), baseline.bounds(),
                    baseline.settlements(), baseline.hive(), baseline.ruleset(), baseline.terrain(), java.util.Map.of(siteId, candidate));
            try {
                FrontierResourceSitePlan.compile(proposed);
                authored = proposed; extended = candidate; break;
            } catch (IllegalArgumentException occupiedOrOutside) {
                // Choose a free neighbour of the authored field, never overwrite another owner.
            }
        }
        assertTrue(authored != null, "fixture needs one owned 65th crop cell next to the field");
        FrontierWorldState state = FrontierWorldRuntimeDefinition.configuration(authored).initialState();
        FrontierWorldState tasked = harvestTask(ready(state), siteId, 22_000L);
        StrategicTask task = onlyHarvestTask(tasked);
        ResourceSiteHarvestStarted started = (ResourceSiteHarvestStarted) ResourceSiteHarvestProcess.plan(tasked,
                ResourceSiteHarvestProcess.start(task, 22_100L)).get(1).payload();
        ResourceSiteHarvestJob job = started.job();
        int previouslyOwned = tasked.inventory().fungibleResources().accounts().get(job.depotAccountId())
                .lotQuantities().values().stream().mapToInt(Integer::intValue).sum();
        assertEquals(65, job.progress().totalCropSlots());
        assertEquals(64, job.segmentEndCropSlot());
        assertEquals(64, job.cropRouteCursors().size());
        FrontierWorldState recovered = new FrontierWorldStateCodec().decode(new FrontierWorldStateCodec().encode(tasked));
        assertEquals(extended, recovered.resourceSite(siteId).layout());
        assertEquals(job, ((ResourceSiteHarvestStarted) ResourceSiteHarvestProcess.plan(recovered,
                ResourceSiteHarvestProcess.start(task, 22_100L)).get(1).payload()).job());
        FrontierWorldStateCodec wrongGenesis = new FrontierWorldStateCodec(baseline);
        assertThrows(IllegalArgumentException.class, () -> wrongGenesis.decode(new FrontierWorldStateCodec().encode(tasked)));

        FrontierWorldState active = StrategicObjectiveProcess.reduceTaskTransition(recovered, new SubjectId("settlement:1"),
                new StrategicTaskTransition(task.id(), StrategicTaskStatus.ACTIVE));
        active = ResourceSiteHarvestProcess.reduceStarted(active, siteId, started);
        active = ResourceSiteHarvestProcess.reducePrepared(active, siteId,
                ((PhysicalIntentPrepared) ResourceSiteHarvestProcess.plan(recovered,
                        ResourceSiteHarvestProcess.start(task, 22_100L)).get(2).payload()).intent());
        for (int completed = 1; completed <= 65; completed++) {
            ResourceSiteHarvestJob current = (ResourceSiteHarvestJob) active.resourceSites().site(siteId).activeWork().orElseThrow();
            active = advanceToCurrentCropStation(active, siteId, current.id());
            active = ResourceSiteHarvestProcess.reduceCropPrepared(active, siteId,
                    new ResourceSiteHarvestCropPrepared(job.id(), completed - 1));
            active = ResourceSiteHarvestProcess.reduceProgressed(active, siteId,
                    progressReceipt(active, siteId, job.id(), completed));
            if (completed == 64) {
                current = (ResourceSiteHarvestJob) active.resourceSites().site(siteId).activeWork().orElseThrow();
                assertTrue(current.returningForBatch());
                active = advanceToDepotWithDueEvidence(active, siteId);
                current = (ResourceSiteHarvestJob) active.resourceSites().site(siteId).activeWork().orElseThrow();
                assertEquals(64, current.deliveredYieldQuantity());
                assertEquals(64, current.segmentStartCropSlot());
                active = new FrontierWorldStateCodec().decode(new FrontierWorldStateCodec().encode(active));
            }
        }
        active = advanceToDepotWithDueEvidence(active, siteId);
        assertEquals(ResourceSitePhase.GROWING, active.resourceSites().site(siteId).phase());
        assertEquals(2L, active.resourceSites().site(siteId).growthEpoch());
        assertEquals(previouslyOwned + 65, active.inventory().fungibleResources().accounts().get(job.depotAccountId())
                .lotQuantities().values().stream().mapToInt(Integer::intValue).sum());

        var unreachableCells = new java.util.ArrayList<>(extended.cells());
        ResourceFieldLayout.Cell first = unreachableCells.getFirst();
        unreachableCells.set(0, new ResourceFieldLayout.Cell(first.id(), first.crop(), first.soil(),
                new SurfaceAnchor(first.workstation().support().offset(0, 100, 0))));
        ResourceFieldLayout unreachable = new ResourceFieldLayout(1L, extended.nextCellId(),
                unreachableCells, extended.irrigationSlots());
        FrontierBootstrap blockedGenesis = new FrontierBootstrap(baseline.worldId(), baseline.seed(), baseline.bounds(),
                baseline.settlements(), baseline.hive(), baseline.ruleset(), baseline.terrain(), java.util.Map.of(siteId, unreachable));
        FrontierWorldState blocked = harvestTask(ready(FrontierWorldState.initial(blockedGenesis)), siteId, 22_000L);
        StrategicTask blockedTask = onlyHarvestTask(blocked);
        assertEquals(List.of(new io.farfrontier.palemirror.frontier.v3.api.ProposedEvent(blockedTask.ownerId(),
                        new StrategicTaskTransition(blockedTask.id(), StrategicTaskStatus.BLOCKED))),
                ResourceSiteHarvestProcess.plan(blocked, ResourceSiteHarvestProcess.start(blockedTask, 22_100L)),
                "an unreachable authored workstation blocks only its own task, without a kernel exception");
    }

    private static FrontierWorldState advanceToDepotWithDueEvidence(FrontierWorldState state, SubjectId siteId) {
        FrontierWorldState current = state;
        while (current.resourceSites().site(siteId).phase() == ResourceSitePhase.HARVESTING) {
            ResourceSiteHarvestJob job = (ResourceSiteHarvestJob) current.resourceSites().site(siteId).activeWork().orElseThrow();
            if (!job.hasNextTraversalStep()) break;
            ScheduledAction due = ResourceSiteHarvestProcess.coldProgress(job, 22_301L);
            current = ResourceSiteHarvestProcess.reduceColdTraversalAdvanced(current, siteId,
                    new ResourceSiteHarvestColdTraversalAdvanced(job.id(), job.workerId(), job.traversalCursor() + 1,
                            due.id(), due.dueAt().ticks()));
            if (job.returningForBatch() && current.resourceSites().site(siteId).activeWork().filter(
                    ResourceSiteHarvestJob.class::isInstance).map(ResourceSiteHarvestJob.class::cast)
                    .map(next -> next.deliveredYieldQuantity() == 64).orElse(false)) break;
        }
        return current;
    }

    @Test
    void admittedHarvestReservesItsExactDepotSlotAcrossRestart() {
        SubjectId site = new SubjectId("site:1-wheat-field");
        FrontierWorldState tasked = harvestTask(ready(initial()), site, 22_000L);
        StrategicTask task = onlyHarvestTask(tasked);
        List<io.farfrontier.palemirror.frontier.v3.api.ProposedEvent> planned = ResourceSiteHarvestProcess.plan(tasked,
                ResourceSiteHarvestProcess.start(task, 22_100L));
        ResourceSiteHarvestJob job = ((ResourceSiteHarvestStarted) planned.get(1).payload()).job();
        assertTrue(tasked.containerSlotAvailable(job.outputSlot()), "the selected slot must really be free before admission");
        assertThrows(IllegalArgumentException.class, () -> new ResourceSiteHarvestJob(job.id(), job.taskId(),
                job.siteId(), job.workerId(), job.actorAccountId(), new SubjectId("custody:foreign-depot"),
                job.outputItemId(), job.outputSlot(), job.intentId(), job.progress(), job.traversal(),
                job.firstCropCursor(), job.traversalCursor()),
                "a forged destination account cannot take the declared chest slot");

        FrontierWorldState active = StrategicObjectiveProcess.reduceTaskTransition(tasked, new SubjectId("settlement:1"),
                (StrategicTaskTransition) planned.getFirst().payload());
        FrontierWorldState harvesting = ResourceSiteHarvestProcess.reduceStarted(active, site,
                (ResourceSiteHarvestStarted) planned.get(1).payload());
        FrontierWorldState preparedHarvesting = ResourceSiteHarvestProcess.reducePrepared(harvesting, site,
                ((PhysicalIntentPrepared) planned.get(2).payload()).intent());
        ExactItemStack colliding = new ExactItemStack(new SubjectId("item:forced-harvest-slot"),
                new SubjectId("settlement:1"), "minecraft:cobblestone", 1, job.outputSlot());
        assertThrows(IllegalArgumentException.class, () -> harvesting.withInventory(harvesting.inventory().store(colliding)),
                "a direct inventory update cannot bypass the active job's output-slot reservation");
        SubjectId accountId = new SubjectId("custody:container-1-depot");
        FungibleResourceLedger resources = harvesting.inventory().fungibleResources();
        var overlapping = FungiblePhysicalObservation.bind(resources, accountId, 1L,
                List.of(new FungiblePhysicalObservation.Stack(new PhysicalStackAddress.ContainerSlot(job.outputSlot()),
                        "minecraft:wheat", 64)));
        FrontierWorldState beforeBinding = harvesting;
        assertThrows(IllegalArgumentException.class, () -> beforeBinding.withInventory(beforeBinding.inventory()
                .withFungibleResources(resources.rebind(accountId, 1L, overlapping))),
                "a HOT fungible binding cannot silently occupy the promised output slot");
        for (FrontierWorldState state : List.of(preparedHarvesting,
                new FrontierWorldStateCodec().decode(new FrontierWorldStateCodec().encode(preparedHarvesting)))) {
            ResourceSiteHarvestJob retained = (ResourceSiteHarvestJob) state.resourceSites().site(site).activeWork().orElseThrow();
            assertEquals(job.actorAccountId(), retained.actorAccountId(),
                    "the declared farmer account must survive snapshot recovery without reconstruction from a job ID");
            assertEquals(job.depotAccountId(), retained.depotAccountId(),
                    "the declared depot account must survive snapshot recovery without reconstruction from a container ID");
            PhysicalIntent retainedIntent = state.physicalIntents().get(job.intentId());
            assertEquals(job.actorAccountId(), retainedIntent.roles().require(PhysicalIntentSubjectRole.CUSTODY_ACCOUNT));
            assertEquals(job.depotAccountId(), retainedIntent.roles().require(PhysicalIntentSubjectRole.RESOURCE_DESTINATION_ACCOUNT));
            assertFalse(retainedIntent.roles().namedRoles().containsKey(PhysicalIntentSubjectRole.OUTPUT_ITEM),
                    "a field intent cannot promise one exact 64-stack when cells can yield zero or partially");
            assertTrue(FrontierDomainRelationships.view(state).edges().stream().anyMatch(edge ->
                    edge.kind() == FrontierDomainRelationships.Kind.HARVEST_ACTOR_ACCOUNT
                            && edge.source().equals(new FrontierDomainRelationships.SubjectEndpoint(
                            FrontierDomainRelationships.EntityKind.RESOURCE_HARVEST_JOB, job.id()))
                            && edge.target().equals(new FrontierDomainRelationships.SubjectEndpoint(
                            FrontierDomainRelationships.EntityKind.RESOURCE_ACCOUNT, job.actorAccountId()))),
                    "the active harvest declares its resource-account relationship before the first yielded cell");
            assertTrue(FrontierDomainRelationships.view(state).edges().stream().anyMatch(edge ->
                    edge.kind() == FrontierDomainRelationships.Kind.HARVEST_DEPOT_ACCOUNT
                            && edge.target().equals(new FrontierDomainRelationships.SubjectEndpoint(
                            FrontierDomainRelationships.EntityKind.RESOURCE_ACCOUNT, job.depotAccountId()))),
                    "the active harvest declares its exact destination resource account");
            assertTrue(state.harvestOutputReserves(job.outputSlot()));
            assertFalse(state.containerSlotAvailable(job.outputSlot()),
                    "another deposit cannot take the output slot before the field yields");
            assertTrue(state.firstFreeContainerSlot(job.outputSlot().containerId()).isPresent());
            assertTrue(state.firstFreeContainerSlot(job.outputSlot().containerId()).orElseThrow() != job.outputSlot().slot(),
                    "another producer must select a different depot slot");
        }
    }

    @Test
    void coldFarmerCannotOvertakeAnUnresolvedPlayerBreakAtItsNextCellAfterRestart() {
        SubjectId site = new SubjectId("site:1-wheat-field");
        FrontierWorldState tasked = harvestTask(ready(initial()), site, 22_000L);
        StrategicTask task = onlyHarvestTask(tasked);
        List<io.farfrontier.palemirror.frontier.v3.api.ProposedEvent> planned = ResourceSiteHarvestProcess.plan(tasked,
                ResourceSiteHarvestProcess.start(task, 22_100L));
        ResourceSiteHarvestStarted started = (ResourceSiteHarvestStarted) planned.get(1).payload();
        FrontierWorldState harvesting = StrategicObjectiveProcess.reduceTaskTransition(tasked, new SubjectId("settlement:1"),
                (StrategicTaskTransition) planned.getFirst().payload());
        harvesting = ResourceSiteHarvestProcess.reduceStarted(harvesting, site, started);
        harvesting = ResourceSiteHarvestProcess.reducePrepared(harvesting, site,
                ((PhysicalIntentPrepared) planned.get(2).payload()).intent());
        ScheduledAction cold = ((ScheduleEffect.Created) planned.get(3).payload()).action();
        ScheduledAction foreignSameSite = new ScheduledAction(
                new io.farfrontier.palemirror.frontier.v3.api.ScheduleId("schedule:resource-site-harvest-cold-progress-foreign"),
                cold.dueAt(), cold.priority(), site, cold.kind(), cold.weight());
        FrontierWorldState admitted = harvesting;
        assertThrows(IllegalArgumentException.class, () -> ResourceSiteHarvestProcess.planColdProgress(admitted, foreignSameSite),
                "a forged job action at the correct site must not vanish as an empty scheduler result");
        ResourceFieldCycle field = harvesting.resourceSites().cycle(site);
        ResourceFieldLayout.CellId nextCell = field.layout().cells().getFirst().id();
        var before = ResourceFieldPhysicalSurface.Condition.of(field.cell(nextCell));
        String actionId = "player:harvest-cold-fence";
        FrontierWorldState pending = ResourceSiteProcess.reducePlayerBreakPrepared(harvesting, site,
                new ResourceFieldPlayerBreakPrepared(site, field.epoch(), field.layout().revision(), nextCell, before,
                        java.util.UUID.fromString("00000000-0000-0000-0000-000000000125"), actionId));
        FrontierWorldState recovered = new FrontierWorldStateCodec().decode(new FrontierWorldStateCodec().encode(pending));

        assertTrue(ResourceSiteHarvestProcess.coldProgressHeld(recovered, cold));
        assertEquals(List.of(new io.farfrontier.palemirror.frontier.v3.api.ProposedEvent(cold.subject(),
                new ScheduleEffect.Rescheduled(cold.id(), cold))),
                ResourceSiteHarvestProcess.planColdProgress(recovered, cold));

        FrontierWorldState closed = ResourceSiteProcess.reduceCellObserved(recovered, site,
                new ResourceFieldCellObserved(site, field.epoch(), field.layout().revision(), nextCell, before, before,
                        ResourceFieldCellObserved.Change.UNCHANGED, ResourceFieldCellObserved.Source.PLAYER, actionId));
        assertFalse(ResourceSiteHarvestProcess.coldProgressHeld(closed, cold));
        assertInstanceOf(ResourceSiteHarvestColdGoalAdvanced.class,
                ResourceSiteHarvestProcess.planColdProgress(closed, cold).getFirst().payload());
    }

    @Test
    void fieldWorkerRetainsOneAdjacentTopologyAndCannotPrepareACropBeforeObservedArrival() {
        FrontierWorldState ready = ready(initial()); SubjectId site = new SubjectId("site:1-wheat-field");
        FrontierWorldState tasked = harvestTask(ready, site, 22_000L); StrategicTask task = onlyHarvestTask(tasked);
        ResourceSiteHarvestStarted started = (ResourceSiteHarvestStarted) ResourceSiteHarvestProcess.plan(tasked,
                ResourceSiteHarvestProcess.start(task, 22_100L)).get(1).payload();
        ResourceSiteHarvestJob job = started.job();
        assertEquals(tasked.actorLocations().get(job.workerId()).supportingSurface(), job.traversal().linearCorridorSurfaces().getFirst());
        assertEquals(ResourceSiteHarvestTraversal.workReturnSurface(tasked.bootstrap(), tasked.resourceSite(site)),
                job.traversal().linearCorridorSurfaces().get(job.firstCropCursor() + job.progress().totalCropSlots() - 1
                        + ResourceSiteHarvestTraversal.workReturnStationCount()),
                "the field-edge tail is retained after all crop cursors before the depot service route");
        for (int index = 1; index < job.traversal().linearCorridorSurfaces().size(); index++) {
            SurfaceAnchor prior = job.traversal().linearCorridorSurfaces().get(index - 1), next = job.traversal().linearCorridorSurfaces().get(index);
            assertEquals(1, Math.abs(prior.x() - next.x()) + Math.abs(prior.z() - next.z()));
            assertTrue(Math.abs(prior.y() - next.y()) <= 1);
        }
        FrontierWorldState active = StrategicObjectiveProcess.reduceTaskTransition(tasked, new SubjectId("settlement:1"),
                new StrategicTaskTransition(task.id(), StrategicTaskStatus.ACTIVE));
        FrontierWorldState harvesting = ResourceSiteHarvestProcess.reduceStarted(active, site, started);
        assertThrows(IllegalArgumentException.class, () -> ResourceSiteHarvestProcess.reduceCropPrepared(harvesting, site,
                new ResourceSiteHarvestCropPrepared(job.id(), 0)));
        FrontierWorldState atField = advanceToCurrentCropStation(harvesting, site, job.id());
        ResourceSiteHarvestJob arrived = (ResourceSiteHarvestJob) atField.resourceSites().site(site).activeWork().orElseThrow();
        assertTrue(arrived.atCurrentCropStation());
        assertEquals(arrived, new FrontierWorldStateCodec().decode(new FrontierWorldStateCodec().encode(atField))
                .resourceSites().site(site).activeWork().orElseThrow(), "restart retains the same field-work cursor and topology");
    }

                    "minecraft:cobblestone", 1, new InventoryCustody.ContainerSlot(depot, slot))));
        }
        return state;
    }

    static FrontierWorldState harvestTask(FrontierWorldState state, SubjectId site, long dueAt) {
        List<io.farfrontier.palemirror.frontier.v3.api.ProposedEvent> planned = StrategicObjectiveProcess.planResourceHarvestOpportunity(state,
                StrategicObjectiveProcess.resourceHarvestOpportunity(state, state.resourceSites().site(site), dueAt));
        assertEquals(3, planned.size());
        assertEquals(planned.getFirst().payload(), FrontierWorldRuntimeDefinition.payloadCodecs().decode(planned.getFirst().payload().type(),
                FrontierWorldRuntimeDefinition.payloadCodecs().encode(planned.getFirst().payload())));
        assertEquals(planned.get(1).payload(), FrontierWorldRuntimeDefinition.payloadCodecs().decode(planned.get(1).payload().type(),
                FrontierWorldRuntimeDefinition.payloadCodecs().encode(planned.get(1).payload())));
        FrontierWorldState selected = StrategicObjectiveProcess.reduceObjective(state, new SubjectId("settlement:1"),
                (StrategicObjectiveSelected) planned.getFirst().payload());
        return StrategicObjectiveProcess.reduceTask(selected, new SubjectId("settlement:1"), (StrategicTaskPlanned) planned.get(1).payload());
    }

    static StrategicTask onlyHarvestTask(FrontierWorldState state) {
        return state.strategicPlans().tasks().values().stream().filter(task -> task.kind() == StrategicTaskKind.HARVEST_RESOURCE_SITE).findFirst().orElseThrow();
    }

    static ResourceSiteHarvestObservation receipt(PhysicalIntent intent, ResourceSiteHarvestJob job, ExactItemStack output) {
        return new ResourceSiteHarvestObservation(new PhysicalObservationId("observation:" + intent.id().value().replace(':', '-')),
                intent.id(), job.siteId(), job.workerId(), output, 64);
    }

    static FrontierWorldState completeHarvestWork(FrontierWorldState state, SubjectId site, ResourceSiteHarvestJob job) {
        FrontierWorldState current = state;
        for (int index = 0; index < ResourceSiteHarvestProgress.TOTAL_CROP_SLOTS; index++) {
            current = advanceToCurrentCropStation(current, site, job.id());
            current = ResourceSiteHarvestProcess.reduceCropPrepared(current, site, new ResourceSiteHarvestCropPrepared(job.id(), index));
            current = ResourceSiteHarvestProcess.reduceProgressed(current, site,
                    progressReceipt(current, site, job.id(), index + 1));
        }
        return current;
    }

    /**
     * Test-only completion path for an already admitted scene.  It deliberately emits the same
     * exact observed HOT checkpoint that a loaded executor must submit instead of smuggling
     * scene-owned progress through the COLD reducer.
     */
    static FrontierWorldState completeHarvestWorkHot(FrontierWorldState state, SubjectId site,
                                                             ResourceSiteHarvestJob job, SceneLeaseId leaseId) {
        FrontierWorldState current = state;
        for (int index = 0; index < ResourceSiteHarvestProgress.TOTAL_CROP_SLOTS; index++) {
            current = advanceToCurrentCropStationHot(current, site, job.id(), leaseId);
            current = ResourceSiteHarvestProcess.reduceCropPrepared(current, site, new ResourceSiteHarvestCropPrepared(job.id(), index));
            ResourceSiteHarvestProgressed receipt = progressReceipt(current, site, job.id(), index + 1);
            SceneLease lease = current.sceneLeases().get(leaseId);
            current = ResourceSiteHarvestProcess.reduceProgressed(current, site,
                    new ResourceSiteHarvestProgressed(receipt.siteId(), receipt.epoch(), receipt.jobId(),
                            receipt.completedCropSlots(), receipt.layoutRevision(), receipt.cellId(), receipt.outcome(),
                            receipt.coldScheduleId(), receipt.coldDueAt(), Optional.of(
                            new ResourceSiteHarvestProgressed.HandObservation(
                                    new PhysicalStackAddress.ActorHand(job.workerId(), lease.members().getFirst().entityId()),
                                    lease.revision(), index + 1))));
        }
        return advanceToCurrentCropStationHot(current, site, job.id(), leaseId);
    }

    static ResourceSiteHarvestProgressed progressReceipt(FrontierWorldState state, SubjectId site,
                                                        SubjectId jobId, int completed) {
        ResourceFieldCycle field = state.resourceSites().cycle(site);
        ResourceFieldLayout.CellId cellId = field.layout().cells().get(completed - 1).id();
        ResourceSiteHarvestJob job = (ResourceSiteHarvestJob) state.resourceSites().site(site).activeWork().orElseThrow();
        ScheduledAction action = ResourceSiteHarvestProcess.coldProgress(job, 22_301L);
        return new ResourceSiteHarvestProgressed(site, field.epoch(), jobId, completed, field.layout().revision(), cellId,
                field.expectedWorkOutcome(cellId), action.id(), action.dueAt().ticks());
    }

    static FrontierWorldState advanceToCurrentCropStation(FrontierWorldState state, SubjectId site, SubjectId jobId) {
        FrontierWorldState current = state;
        ResourceSiteHarvestJob job = (ResourceSiteHarvestJob) current.resourceSites().site(site).activeWork().orElseThrow();
        while (job.hasNextTraversalStep()) {
            current = ResourceSiteHarvestProcess.reduceColdTraversalAdvanced(current, site,
                    new ResourceSiteHarvestColdTraversalAdvanced(jobId, job.workerId(), job.traversalCursor() + 1));
            job = (ResourceSiteHarvestJob) current.resourceSites().site(site).activeWork().orElseThrow();
        }
        return current;
    }

    static FrontierWorldState advanceToCurrentCropStationHot(FrontierWorldState state, SubjectId site,
                                                                       SubjectId jobId, SceneLeaseId leaseId) {
        FrontierWorldState current = state;
        ResourceSiteHarvestJob job = (ResourceSiteHarvestJob) current.resourceSites().site(site).activeWork().orElseThrow();
        while (job.hasNextTraversalStep()) {
            BodyPosition observed = job.nextTraversalSurface().standingBody();
            current = ResourceSiteHarvestProcess.reduceHotTraversalAdvanced(current, site,
                    new ResourceSiteHarvestHotTraversalAdvanced(jobId, leaseId, job.workerId(), observed,
                            job.traversalCursor() + 1));
            job = (ResourceSiteHarvestJob) current.resourceSites().site(site).activeWork().orElseThrow();
        }
        return current;
    }

    static HotHarvest hotHarvestAfterColdSteps(int coldSteps) {
        return hotHarvestAfterColdSteps(125L, coldSteps);
    }

    static HotHarvest hotHarvestAfterColdSteps(long seed, int coldSteps) {
        ColdHarvest cold = coldHarvestAfterSteps(seed, coldSteps);
        FrontierWorldState state = cold.state();
        SubjectId site = cold.site();
        ResourceSiteHarvestJob job = cold.job();
        SceneLease lease = newHarvestLease(state, site, job, "hot-checkpoint-" + seed + "-" + coldSteps);
        state = state.prepareSceneLease(lease).transitionSceneLease(lease.id(), SceneLeaseStatus.HOT);
        return new HotHarvest(state, site, job, state.sceneLeases().get(lease.id()));
    }

    static ColdHarvest coldHarvestAfterSteps(long seed, int coldSteps) {
        FrontierWorldState ready = ready(initial(seed));
        SubjectId site = new SubjectId("site:1-wheat-field");
        FrontierWorldState tasked = harvestTask(ready, site, 22_000L);
        StrategicTask task = onlyHarvestTask(tasked);
        List<io.farfrontier.palemirror.frontier.v3.api.ProposedEvent> planned = ResourceSiteHarvestProcess.plan(tasked,
                ResourceSiteHarvestProcess.start(task, 22_100L));
        ResourceSiteHarvestStarted started = (ResourceSiteHarvestStarted) planned.get(1).payload();
        FrontierWorldState state = StrategicObjectiveProcess.reduceTaskTransition(tasked, new SubjectId("settlement:1"),
                (StrategicTaskTransition) planned.getFirst().payload());
        state = ResourceSiteHarvestProcess.reduceStarted(state, site, started);
        state = ResourceSiteHarvestProcess.reducePrepared(state, site, ((PhysicalIntentPrepared) planned.get(2).payload()).intent());
        for (int index = 0; index < coldSteps; index++) {
            ResourceSiteHarvestJob current = (ResourceSiteHarvestJob) state.resourceSites().site(site).activeWork().orElseThrow();
            state = ResourceSiteHarvestProcess.reduceColdTraversalAdvanced(state, site,
                    new ResourceSiteHarvestColdTraversalAdvanced(current.id(), current.workerId(), current.traversalCursor() + 1));
        }
        return new ColdHarvest(state, site, (ResourceSiteHarvestJob) state.resourceSites().site(site).activeWork().orElseThrow());
    }

    static SceneLease newHarvestLease(FrontierWorldState state, SubjectId site, ResourceSiteHarvestJob job, String suffix) {
        BodyPosition current = job.traversal().linearCorridorSurfaces().get(job.traversalCursor()).standingBody();
        BlockPosition crop = FrontierResourceSitePlan.compile(state.bootstrap()).get(site).cropSlots().get(job.progress().nextCropSlotIndex());
        return SceneLease.forCause(new SceneLeaseId("lease:site-harvest-" + suffix), state.bootstrap().worldId(),
                new ResourceSiteHarvestSceneCause(job.siteId(), job.id()), crop, new SimInstant(22_300L), 1L, SceneLeaseStatus.PREPARED,
                List.of(new SceneMember(job.workerId(), SceneLease.deterministicEntityId(state.bootstrap().worldId(), job.workerId()))),
                java.util.Map.of(job.workerId(), current), java.util.Set.of(job.workerId()), Optional.empty());
    }

    static CommandPlan hotCheckpointPlan(FrontierWorldState state, String suffix,
                                                  ResourceSiteHarvestHotTraversalAdvanced checkpoint) {
        CommandId commandId = new CommandId("command:site-harvest-hot-" + suffix);
        SceneLease actualLease = Optional.ofNullable(state.sceneLeases().get(checkpoint.leaseId()))
                .orElseGet(() -> state.sceneLeases().values().stream()
                        .filter(FrontierSceneBehaviors::isResourceSiteHarvest).findFirst().orElseThrow());
        ResourceSiteHarvestSceneCause cause = FrontierSceneBehaviors.resourceSiteHarvest(actualLease);
        ResourceSiteHarvestJob job = FrontierResourceSiteHarvestSceneSupport.require(state, cause);
        ScheduledAction action = ResourceSiteHarvestProcess.coldProgress(job, 22_301L);
        FrontierCommand command = new FrontierCommand(FrontierCommand.SCHEMA_VERSION, commandId, state.bootstrap().worldId(), new Revision(1L),
                new SimInstant(22_301L), FrontierWorldRuntimeDefinition.PHYSICAL_EXECUTOR, CauseChain.root(commandId), checkpoint,
                Optional.of(new EngineScheduleBinding(new Revision(1L), action)));
        return FrontierWorldProcessCatalog.planCommand("resource-sites", state, command);
    }

    static void assertRejectedHotCheckpoint(FrontierWorldState state, String suffix,
                                                    ResourceSiteHarvestHotTraversalAdvanced checkpoint) {
        assertTrue(hotCheckpointPlan(state, suffix, checkpoint) instanceof CommandPlan.Rejected,
                "invalid HOT checkpoint must be rejected before event persistence: " + suffix);
    }

    static final class CalibrationFacts {
        private final LinkedHashSet<String> actors = new LinkedHashSet<>(), objects = new LinkedHashSet<>(), stages = new LinkedHashSet<>();
        private final LinkedHashSet<String> topology = new LinkedHashSet<>(), opportunities = new LinkedHashSet<>();
        private final LinkedHashMap<String, Long> claims = new LinkedHashMap<>(), custody = new LinkedHashMap<>(), work = new LinkedHashMap<>();
        private final LinkedHashMap<String, String> recovery = new LinkedHashMap<>();

        void append(FrontierWorldState state, ResourceSiteHarvestJob job, SubjectId site, long seed) {
            String prefix = "seed:" + seed + ":";
            actors.add(prefix + job.workerId().value());
            objects.add(prefix + job.id().value());
            claims.put(prefix + "worker", 1L);
            custody.put(prefix + "crop-slots", (long) job.progress().completedCropSlots());
            custody.put(prefix + "body-x", 4_096L + state.actorLocations().get(job.workerId()).body().x());
            custody.put(prefix + "body-y", 4_096L + state.actorLocations().get(job.workerId()).body().y());
            custody.put(prefix + "body-z", 4_096L + state.actorLocations().get(job.workerId()).body().z());
            stages.add(prefix + state.resourceSites().site(site).phase());
            work.put(prefix + "cursor", (long) job.traversalCursor());
            topology.add(prefix + job.traversal().linearCorridorSurfaces());
            recovery.put(prefix + "intent", state.physicalIntents().get(job.intentId()).status().name());
            recovery.put(prefix + "active-lease", "none");
            opportunities.add(prefix + job.id().value());
        }

        FrontierObserverNeutralityContract.Run run(FrontierObserverNeutralityContract.Declaration declaration) {
            int samples = actors.size();
            return new FrontierObserverNeutralityContract.Run(declaration, actors, objects, claims, custody, stages, work, topology,
                    java.util.Set.of(), recovery, opportunities,
                    new FrontierObserverNeutralityContract.CalibrationSample(samples, samples, 0, samples * 400L));
        }
    }

    static FrontierWorldState initial() {
        return FrontierWorldState.initial(FrontierBootstrapper.create(new WorldId("frontier:resource-site-harvest"), 125L));
    }

    static FrontierWorldState initial(long seed) {
        return FrontierWorldState.initial(FrontierBootstrapper.create(new WorldId("frontier:resource-site-harvest-" + seed), seed));
    }

    static long column(int x, int z) {
        return (Integer.toUnsignedLong(x) << 32) | Integer.toUnsignedLong(z);
    }
